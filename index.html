    <script>

        // Search Filter Function

        function filterResources() {

            let input = document.getElementById('searchInput').value.toLowerCase();

            let cards = document.querySelectorAll('.resource-card');

            let activeCategory = document.querySelector('.tab-btn.bg-indigo-600').getAttribute('onclick').match(/'([^']+)'/)[1];

            let visibleCount = 0;


            cards.forEach(card => {

                let title = card.getAttribute('data-title');

                let category = card.getAttribute('data-category');

                let matchesSearch = title.includes(input);

                let matchesCategory = (activeCategory === 'all' || category === activeCategory);


                if (matchesSearch && matchesCategory) {

                    card.style.display = "flex";

                    visibleCount++;

                } else {

                    card.style.display = "none";

                }

            });


            document.getElementById('noResults').style.display = visibleCount === 0 ? "block" : "none";

        }


        // Category Tab Filter Function

        function filterCategory(category) {

            // Update active styles on tab buttons

            document.querySelectorAll('.tab-btn').forEach(btn => {

                btn.classList.remove('bg-indigo-600', 'text-white', 'shadow');

                btn.classList.add('bg-white', 'text-slate-700', 'border', 'border-slate-200');

            });

            event.currentTarget.classList.remove('bg-white', 'text-slate-700', 'border', 'border-slate-200');

            event.currentTarget.classList.add('bg-indigo-600', 'text-white', 'shadow');


            // Filter cards

            let cards = document.querySelectorAll('.resource-card');

            let input = document.getElementById('searchInput').value.toLowerCase();

            let visibleCount = 0;


            cards.forEach(card => {

                let title = card.getAttribute('data-title');

                let cardCat = card.getAttribute('data-category');

                let matchesSearch = title.includes(input);

                let matchesCategory = (category === 'all' || cardCat === category);


                if (matchesSearch && matchesCategory) {

                    card.style.display = "flex";

                    visibleCount++;

                } else {

                    card.style.display = "none";

                }

            });


            document.getElementById('noResults').style.display = visibleCount === 0 ? "block" : "none";

        }


        // Modal Controls

        function openRequestModal() {

            document.getElementById('requestModal').style.display = 'flex';

        }


        function closeRequestModal() {

            document.getElementById('requestModal').style.display = 'none';

        }


        function handleRequestSubmit(e) {

            e.preventDefault();

            alert('Thank you! Your note request has been successfully recorded.');

            closeRequestModal();

            e.target.reset();

        }

    </script>
